<body>
    <h2>JavaScript Increment and Decrement</h2>

    <script>
        let a = 10;

        document.write("Initial value of a: " + a + "<br>");

        a++;
        document.write("After a++: " + a + "<br>");

        a--;
        document.write("After a--: " + a + "<br>");

        ++a;
        document.write("After ++a: " + a + "<br>");

        --a;
        document.write("After --a: " + a + "<br>");
    </script>
</body>
